# Define function to compute Term Frequency
def computeTF(wordDict, doc):

    # Create empty dictionary for TF values
    tfDict = {}

    # Calculate total number of words in document
    corpusCount = len(doc)

    # Calculate TF value for each word
    for word, count in wordDict.items():

        # Compute TF using formula:
        # TF = Word Count / Total Words
        tfDict[word] = count / float(corpusCount)

    # Return TF dictionary
    return tfDict

# Define function to compute Inverse Document Frequency
def computeIDF(docList):

    # Create empty dictionary for IDF values
    idfDict = {}

    # Calculate total number of documents
    N = len(docList) 
    
    # Collect all unique words from all documents     
    all_words = set()     

    # Traverse each document
    for doc in docList:

        # Add words into all_words set
        all_words.update(doc.keys())
    
    # Initialize IDF dictionary with all unique words     
    idfDict = dict.fromkeys(all_words, 0)
        
    # Compute document frequency for each word     
    for doc in docList:         

        # Traverse each word in document
        for word in doc:             

            # Check whether word exists in document
            if doc[word] > 0:                 

                # Increase document frequency count
                idfDict[word] += 1      

    # Compute IDF values     
    for word in idfDict:         

        # Apply IDF formula
        idfDict[word] = math.log10(
            N / (float(idfDict[word]) + 1)
        )      

    # Return IDF dictionary
    return idfDict

# Define function to compute TF-IDF values
def computeTFIDF(tfBow, idfs):

    # Create empty dictionary for TF-IDF
    tfidf = {}     

    # Traverse each word and TF value
    for word, val in tfBow.items():         

        # Calculate TF-IDF value
        tfidf[word] = val * idfs.get(word, 0)     

    # Return TF-IDF dictionary
    return tfidf
