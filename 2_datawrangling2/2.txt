# OUTLIER DETECTION AND TREATMENT
# ============================================================


# ------------------------------------------------------------
# STEP 22: IMPORT REQUIRED LIBRARIES
# ------------------------------------------------------------

# Import pandas library
import pandas as pd

# Import numpy library
import numpy as np


# ------------------------------------------------------------
# STEP 23: LOAD SECOND DATASET
# ------------------------------------------------------------

# Read dataset containing outliers
df = pd.read_csv("student_performance1.csv")


# ------------------------------------------------------------
# STEP 24: DISPLAY DATASET
# ------------------------------------------------------------

# Display dataset
df


# ------------------------------------------------------------
# STEP 25: IMPORT MATPLOTLIB
# ------------------------------------------------------------

# Import matplotlib for plotting graphs
import matplotlib.pyplot as plt


# ------------------------------------------------------------
# STEP 26: SELECT NUMERICAL COLUMNS
# ------------------------------------------------------------

# Select numerical columns for analysis
box_df = df[[
    'math score',
    'reading score',
    'writing score',
    'placement score'
]]


# ------------------------------------------------------------
# STEP 27: CREATE BOXPLOT
# ------------------------------------------------------------

# Create boxplot for numerical columns
ax = box_df.plot(kind='box')

# Display graph
plt.show()


# ------------------------------------------------------------
# STEP 28: DETECT OUTLIERS USING CONDITIONS
# ------------------------------------------------------------

# Display indexes where math score is greater than 90
print(np.where(df['math score'] > 90))

# Display indexes where reading score is less than 25
print(np.where(df['reading score'] < 25))

# Display indexes where writing score is less than 30
print(np.where(df['writing score'] < 30))


# ------------------------------------------------------------
# STEP 29: SORT READING SCORES
# ------------------------------------------------------------

# Sort reading score values
sorted_rscore = sorted(df['reading score'])

# Display sorted values
sorted_rscore


# ------------------------------------------------------------
# STEP 30: CALCULATE QUARTILES
# ------------------------------------------------------------

# Calculate first quartile (Q1)
q1 = np.percentile(sorted_rscore, 25)

# Calculate third quartile (Q3)
q3 = np.percentile(sorted_rscore, 75)

# Display quartile values
print(q1, q3)


# ------------------------------------------------------------
# STEP 31: CALCULATE INTERQUARTILE RANGE
# ------------------------------------------------------------

# Calculate Interquartile Range
IQR = q3 - q1


# ------------------------------------------------------------
# STEP 32: CALCULATE LOWER AND UPPER BOUNDS
# ------------------------------------------------------------

# Calculate lower bound for outliers
lwr_bound = q1 - (1.5 * IQR)

# Calculate upper bound for outliers
upr_bound = q3 + (1.5 * IQR)

# Display bounds
print(lwr_bound, upr_bound)


# ------------------------------------------------------------
# STEP 33: DETECT OUTLIERS USING IQR METHOD
# ------------------------------------------------------------

# Create empty list for outliers
r_outliers = []

# Traverse all reading scores
for i in sorted_rscore:

    # Check whether value is outlier
    if (i < lwr_bound or i > upr_bound):

        # Append outlier value
        r_outliers.append(i)

# Display outlier values
print(r_outliers)


# ------------------------------------------------------------
# STEP 34: CREATE BOXPLOT FOR READING SCORE
# ------------------------------------------------------------

# Select reading score column
col = ['reading score']

# Create boxplot
df.boxplot(column=col)

# Display graph
plt.show()


# ------------------------------------------------------------
# STEP 35: CALCULATE MEDIAN
# ------------------------------------------------------------

# Calculate median of reading scores
median = np.median(sorted_rscore)

# Display median value
median


# ------------------------------------------------------------
# STEP 36: TREAT OUTLIERS USING MEDIAN REPLACEMENT
# ------------------------------------------------------------

# Import matplotlib library
import matplotlib.pyplot as plt

# Create refined dataframe
refined_df = df

# Replace upper outliers with median
refined_df['reading score'] = np.where(
    refined_df['reading score'] > upr_bound,
    median,
    refined_df['reading score']
)

# Replace lower outliers with median
refined_df['reading score'] = np.where(
    refined_df['reading score'] < lwr_bound,
    median,
    refined_df['reading score']
)

# Select reading score column
col = ['reading score']

# Create boxplot after removing outliers
refined_df.boxplot(column=col)

# Display graph
plt.show()
