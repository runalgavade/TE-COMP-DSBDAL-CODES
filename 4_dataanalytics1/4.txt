# ------------------------------------------------------------
# STEP 1: IMPORT REQUIRED LIBRARIES
# ------------------------------------------------------------

# Import libraries and create alias for Pandas, Numpy and Matplotlib

# Import pandas library for data handling
import pandas as pd

# Import numpy library for numerical calculations
import numpy as np

# Import matplotlib library for plotting graphs
import matplotlib.pyplot as plt

# Import warnings library
import warnings

# Ignore warning messages
warnings.filterwarnings('ignore')

# Ignore simple warnings
warnings.simplefilter('ignore')


# ------------------------------------------------------------
# STEP 2: CREATE SIMPLE LINEAR REGRESSION DATA
# ------------------------------------------------------------

# Create a Dataframe with Dependent Variable(x) and independent variable y.

# Create input feature array x
x=np.array([95,85,80,70,60])

# Create target output array y
y=np.array([85,95,70,65,70])


# ------------------------------------------------------------
# STEP 3: CREATE LINEAR REGRESSION MODEL USING POLYFIT
# ------------------------------------------------------------

# Create Linear Regression Model using Polyfit Function:

# Create linear regression model using numpy polyfit
model= np.polyfit(x, y, 1)

# Observe the coefficients of the model.

# Display slope and intercept values
model


# ------------------------------------------------------------
# STEP 4: PREDICT SINGLE VALUE
# ------------------------------------------------------------

# Predict the Y value for X and observe the output.

# Create prediction function from model
predict = np.poly1d(model)

# Predict y value for x = 65
predict(65)


# ------------------------------------------------------------
# STEP 5: PREDICT ALL VALUES
# ------------------------------------------------------------

# Predict the y_pred for all values of x.

# Predict output values for all x values
y_pred= predict(x)

# Display predicted values
y_pred


# ------------------------------------------------------------
# STEP 6: EVALUATE MODEL USING R2 SCORE
# ------------------------------------------------------------

# Evaluate the performance of Model (R-Suare)

# Import R2 score evaluation metric
from sklearn.metrics import r2_score

# Calculate R2 score of model
r2_score(y, y_pred)


# ------------------------------------------------------------
# STEP 7: PLOT SIMPLE LINEAR REGRESSION MODEL
# ------------------------------------------------------------

# Step 8: Plotting the linear regression model

# Create regression line equation
y_line = model[1] + model[0]* x

# Plot regression line
plt.plot(x, y_line, c = 'r')

# Plot predicted points
plt.scatter(x, y_pred)

# Plot actual points
plt.scatter(x,y,c='r')


# ------------------------------------------------------------
# STEP 8: IMPORT LIBRARIES AGAIN FOR BOSTON DATASET
# ------------------------------------------------------------

#Importing libraries

# Import numpy library
import numpy as np

# Import pandas library
import pandas as pd

# Import matplotlib library
import matplotlib.pyplot as plt


# ------------------------------------------------------------
# STEP 9: LOAD BOSTON HOUSING DATASET
# ------------------------------------------------------------

# Import the Boston Housing dataset

# Initialize the data frame

# Read Boston housing CSV dataset
data = pd.read_csv('boston.csv')

# Convert dataset into dataframe
data = pd.DataFrame(data)

# Remove unnamed columns from dataset
data = data.loc[:, ~data.columns.str.contains('^Unnamed')]


# ------------------------------------------------------------
# STEP 10: DISPLAY FIRST 5 ROWS
# ------------------------------------------------------------

# Display first 5 rows of dataset
data.head()


# ------------------------------------------------------------
# STEP 11: CHECK MISSING VALUES
# ------------------------------------------------------------

# Perform Data Preprocessing( Check for missing values)

# Check total missing values in dataset
data.isnull().sum()


# ------------------------------------------------------------
# STEP 12: SEPARATE FEATURES AND TARGET
# ------------------------------------------------------------

# Separate independent variables from dataset
x = data.drop(['Price'], axis = 1)

# Separate dependent variable Price
y = data['Price']


# ------------------------------------------------------------
# STEP 13: SPLIT DATASET INTO TRAINING AND TESTING DATA
# ------------------------------------------------------------

 # splitting datato trainingandtestingdataset.

# Import train_test_split function
from sklearn.model_selection import train_test_split

# Split dataset into training and testing data
xtrain, xtest,ytrain,ytest=train_test_split(x,y,test_size=0.2,random_state =0)


# ------------------------------------------------------------
# STEP 14: TRAIN LINEAR REGRESSION MODEL
# ------------------------------------------------------------

#Uselinearregression(TraintheMachine)toCreateModel

# Import sklearn library
import sklearn

# Import LinearRegression model
from sklearn.linear_model import LinearRegression

# Create Linear Regression object
lm =LinearRegression()

# Train model using training data
model=lm.fit(xtrain,ytrain)


# ------------------------------------------------------------
# STEP 15: PREDICT TRAINING AND TESTING VALUES
# ------------------------------------------------------------

 #Predict they_pred forallvaluesoftrain_xandtest_x

# Predict values for training dataset
ytrain_pred= lm.predict(xtrain)

# Predict values for testing dataset
ytest_pred= lm.predict(xtest)


# ------------------------------------------------------------
# STEP 16: COMPARE ACTUAL AND PREDICTED VALUES
# ------------------------------------------------------------

 #EvaluatetheperformanceofModelfortrain_yandtest_y

# Create dataframe for training prediction comparison
df=pd.DataFrame(ytrain_pred,ytrain)

# Create dataframe for testing prediction comparison
df=pd.DataFrame(ytest_pred,ytest)


# ------------------------------------------------------------
# STEP 17: CALCULATE MEAN SQUARED ERROR
# ------------------------------------------------------------

#Calculate Mean Square Paperfortrain_yandtest_y

# Import evaluation metrics
from sklearn.metrics import mean_squared_error,r2_score

# Calculate Mean Squared Error for testing data
mse= mean_squared_error(ytest,ytest_pred)

# Display testing MSE
print(mse)

# Calculate Mean Squared Error for training data
mse= mean_squared_error(ytrain_pred,ytrain)

# Display training MSE
print(mse)


# ------------------------------------------------------------
# STEP 18: DISPLAY MSE AGAIN
# ------------------------------------------------------------

# Calculate Mean Squared Error again for testing data
mse= mean_squared_error(ytest,ytest_pred)

# Display testing MSE
print(mse)


# ------------------------------------------------------------
# STEP 19: PLOT FINAL LINEAR REGRESSION GRAPH
# ------------------------------------------------------------

# Plotting the linear regressionmodel

# Plot training data predictions
plt.scatter(ytrain,ytrain_pred,c='blue',marker='o',label='Training data')

# Plot testing data predictions
plt.scatter(ytest,ytest_pred,c='lightgreen',marker='s',label='Test data')

# Set x-axis label
plt.xlabel('True values')

# Set y-axis label
plt.ylabel('Predicted')

# Set graph title
plt.title("True value vs Predictedvalue")

# Display legend on graph
plt.legend(loc='upper left')
